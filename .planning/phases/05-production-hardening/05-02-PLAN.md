**docker-compose.prod.yml** — create as an override file (NOT a full copy):

```yaml
version: "3.9"

# Production overlay — use with: docker compose -f docker-compose.yml -f docker-compose.prod.yml --compatibility up -d
# Only specifies deltas from the base docker-compose.yml.

services:
  langgraph-agent:
    restart: always
    environment:
      APP_ENV: prod
      LOG_FORMAT: json
      MLFLOW_ENABLED: "false"
    deploy:
      resources:
        limits:
          cpus: "4.0"
          memory: 4G
        reservations:
          cpus: "1.0"
          memory: 2G

  qdrant:
    restart: always
    deploy:
      resources:
        limits:
          cpus: "2.0"
          memory: 2G
        reservations:
          memory: 1G

  neo4j:
    restart: always
    deploy:
      resources:
        limits:
          cpus: "2.0"
          memory: 2G
        reservations:
          memory: 1G

  redis:
    restart: always
    deploy:
      resources:
        limits:
          cpus: "1.0"
          memory: 512M
        reservations:
          memory: 256M

  postgres:
    restart: always
    deploy:
      resources:
        limits:
          cpus: "1.0"
          memory: 1G
        reservations:
          memory: 512M

  minio:
    restart: always
    deploy:
      resources:
        limits:
          cpus: "1.0"
          memory: 1G
        reservations:
          memory: 512M

  mlflow:
    restart: always
    deploy:
      resources:
        limits:
          cpus: "1.0"
          memory: 1G
        reservations:
          memory: 512M
```

Key decisions:
- langgraph-agent gets most resources (4 CPU, 4G RAM) — it runs cross-encoder reranking and orchestrates all agents
- LOG_FORMAT=json for structured production logging (existing logging_config.py supports this)
- MLFLOW_ENABLED=false in prod overlay — MLflow adds overhead; enable explicitly when needed
- All services get restart: always (base has unless-stopped)
- Resource reservations ensure minimum availability
