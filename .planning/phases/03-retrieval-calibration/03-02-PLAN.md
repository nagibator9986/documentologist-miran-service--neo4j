   ```python
   def _score_stats(scores: list[float]) -> dict:
       """Return min/max/p50 for a list of scores. Returns empty dict if no scores."""
       if not scores:
           return {}
       return {
           "min": round(min(scores), 4),
           "max": round(max(scores), 4),
           "p50": round(_statistics.median(scores), 4),
       }
   ```
